const b=resolve(process.argv[2]),seed=JSON.parse(readFileSync(b+'/outputs/unicode-project-identity-matrix-input-v1.json')),config=JSON.parse(readFileSync(seed.configPath)),{parse}=createRequire(b+'/../connectors/package.json')('yaml'),records=[];
